employee1 = ("John", 28, "IT", 50000)
employee2 = ("Alice", 32, "HR", 60000)
employee3 = ("Bob", 25, "Marketing", 45000)
employee4 = ("Eve", 30, "Finance", 55000)

employees = (employee1, employee2, employee3, employee4)

print("All Employee Details:")
for emp in employees:
    print(emp)

print("\nAccessing individual employee details:")
print("Name of first employee:", employees[0][0])
print("Department of third employee:", employees[2][2])

print("\nCount of employee 'Bob' tuple:", employees.count(employee3))

print("Index of Alice:", employees.index(employee2))

print("Number of employees:", len(employees))

print("\nAccessing Salary of second employee:", employees[1][3])

new_employee = ("David", 27, "IT", 48000)
all_employees = employees + (new_employee,)
print("\nAfter adding a new employee:")
for emp in all_employees:
    print(emp)

print("\nSlicing first two employees:")
print(all_employees[0:2])
