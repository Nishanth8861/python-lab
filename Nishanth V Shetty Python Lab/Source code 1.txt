a = 10              # Integer
b = 3.5             # Float
c = 2 + 3j          # Complex
name = "Python"     # String
my_list = [1, 2, 3] # List
my_tuple = (4, 5, 6)# Tuple
my_set = {7, 8, 9}  # Set
my_dict = {"name": "John", "age": 21} # Dictionary
flag = True         # Boolean

print("Integer:", a)
print("Float:", b)
print("Complex:", c)
print("String:", name)
print("List:", my_list)
print("Tuple:", my_tuple)
print("Set:", my_set)
print("Dictionary:", my_dict)
print("Boolean:", flag)

print("\nArithmetic Operators")
print("Addition:", a + b)
print("Subtraction:", a - b)
print("Multiplication:", a * b)
print("Division:", a / b)
print("Modulus:", a % 3)
print("Exponent:", a ** 2)
print("Floor Division:", a // 3)

print("\nComparison Operators")
print("a > b:", a > b)
print("a < b:", a < b)
print("a == b:", a == b)
print("a != b:", a != b)

print("\nLogical Operators")
x = True
y = False
print("x and y:", x and y)
print("x or y:", x or y)
print("not x:", not x)

print("\nAssignment Operators")
x = 5
print("Initial x:", x)
x += 3
print("x += 3:", x)
x *= 2
print("x *= 2:", x)


print("p is q:", p is q)
print("p is r:", p is r)
print("p is not r:", p is not r)
