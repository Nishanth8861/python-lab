phone_book = {
    "John": "9876543210",
    "Alice": "9123456780",
    "Bob": "9988776655"
}

print("Phone Book Contacts:")
for name, number in phone_book.items():
    print(f"{name}: {number}")

phone_book["Eve"] = "9012345678"
print("\nAfter adding Eve:")
print(phone_book)

phone_book["John"] = "9999999999"
print("\nAfter updating John's number:")
print(phone_book)

search_name = "Alice"
if search_name in phone_book:
    print(f"\n{search_name}'s number is {phone_book[search_name]}")
else:
    print(f"\n{search_name} not found in phone book")

del phone_book["Bob"]
print("\nAfter deleting Bob:")
print(phone_book)

print("\nUsing get() to find Charlie:", phone_book.get("Charlie", "Not Found"))

print("\nAll names:", list(phone_book.keys()))
print("All numbers:", list(phone_book.values()))
